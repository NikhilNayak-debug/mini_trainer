# Sequential OSFT Training Commands for TRACE Benchmark
# LLaMA-2-7B-Chat with TRACE hyperparameters + OSFT
# Learning Rate: 1e-5, Batch Size: 128, Weight Decay: 0
# Progressive OSFT unfreeze rank ratios

# Run these commands from /workspace/mini_trainer directory
cd /workspace/mini_trainer

# STEP 1: C-STANCE (1 epoch, ~5000 samples) - Base training, no OSFT
torchrun --nnodes=1 --nproc-per-node=8 src/mini_trainer/train.py \
  --output-dir ./trace_outputs/sft_step1_cstance \
  --data-path ./trace_datasets/tokenized/C-STANCE_train_tokenized.jsonl \
  --model-name-or-path meta-llama/Llama-2-7b-chat-hf \
  --learning-rate 1e-5 \
  --batch-size 128 \
  --max-tokens-per-gpu 128000 \
  --max-epochs 1 \
  --use-liger-kernels \
  --save-final-checkpoint

# STEP 2: FOMC (1 epoch, ~5000 samples)
torchrun --nnodes=1 --nproc-per-node=8 src/mini_trainer/train.py \

  --data-path ./trace_datasets/tokenized/FOMC_train_tokenized.jsonl \
  --model-name-or-path ./trace_outputs/sft_step1_cstance/hf_format/samples_5000.0 \
  --learning-rate 5e-6 \
  --batch-size 128 \
  --max-tokens-per-gpu 128000 \
  --max-epochs 1 \
  --save-final-checkpoint \
  --osft \
  --osft-unfreeze-rank-ratio 0.85 \
  --osft-memory-efficient-init
